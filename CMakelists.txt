cmake_minimum_required(VERSION 3.20)
project(MyProject LANGUAGES C CXX)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}")

# =========================
# Compiler settings
# =========================
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -g -O0")

# =========================
# Library directories
# =========================
set(LIB_DIR "../myLibraries")

# -------------------------
# SDL libraries (shared)
# -------------------------
add_library(SDL3 SHARED IMPORTED)
set_target_properties(SDL3 PROPERTIES
    IMPORTED_IMPLIB "${LIB_DIR}/lib/libSDL3.dll.a"
    IMPORTED_LOCATION "${LIB_DIR}/lib/SDL3.dll"
    INTERFACE_INCLUDE_DIRECTORIES "${LIB_DIR}/include"
)

add_library(SDL3_ttf SHARED IMPORTED)
set_target_properties(SDL3_ttf PROPERTIES
    IMPORTED_IMPLIB "${LIB_DIR}/lib/libSDL3_ttf.dll.a"
    IMPORTED_LOCATION "${LIB_DIR}/lib/SDL3_ttf.dll"
    INTERFACE_INCLUDE_DIRECTORIES "${LIB_DIR}/include"
)

add_library(SDL3_image SHARED IMPORTED)
set_target_properties(SDL3_image PROPERTIES
    IMPORTED_IMPLIB "${LIB_DIR}/lib/libSDL3_image.dll.a"
    IMPORTED_LOCATION "${LIB_DIR}/lib/SDL3_image.dll"
    INTERFACE_INCLUDE_DIRECTORIES "${LIB_DIR}/include"
)

add_library(glad STATIC IMPORTED)
set_target_properties(glad PROPERTIES
    IMPORTED_LOCATION "${LIB_DIR}/libraries/libglad.a"
    INTERFACE_INCLUDE_DIRECTORIES "${LIB_DIR}"
)

# -------------------------
# ENet library (static)
# -------------------------
add_library(enet64 STATIC IMPORTED)
set_target_properties(enet64 PROPERTIES
    IMPORTED_LOCATION "${LIB_DIR}/enet-1.3.18/libenet.a"
    INTERFACE_INCLUDE_DIRECTORIES "${LIB_DIR}/enet-1.3.18/include"
)

# -------------------------
# My precompiled library
# -------------------------
add_library(myLibrary STATIC IMPORTED)
set_target_properties(myLibrary PROPERTIES
    IMPORTED_LOCATION "${LIB_DIR}/libraries/libmyLibrary.a"
    INTERFACE_INCLUDE_DIRECTORIES "${LIB_DIR}/libraries/header"
)

# =========================
# Common includes and libraries
# =========================
set(COMMON_INCLUDES
    "./libraries/header"
)

set(COMMON_LIBS
    myLibrary
    glad
    SDL3
    SDL3_ttf
    SDL3_image
)

set(ENET_LIBS
    enet64
    ws2_32
    winmm
)

# =========================
# Executable sources
# =========================

set(NETWORKING_SRC
    "./libraries/src/networking.cpp"
)

set(RENDERING_SRC
    "./libraries/src/render.cpp"
)

set(STATE_SRC
    "./libraries/src/gameState.cpp"
)

# =========================
# Targets
# =========================

# -------- editor --------
add_executable(editor editor.cpp)
target_include_directories(editor PRIVATE ${COMMON_INCLUDES})
target_link_libraries(editor PRIVATE ${COMMON_LIBS})

# -------- raycast --------
add_executable(raycast raycast.cpp ${NETWORKING_SRC} ${RENDERING_SRC} ${STATE_SRC})
target_include_directories(raycast PRIVATE ${COMMON_INCLUDES})
target_link_libraries(raycast PRIVATE ${COMMON_LIBS} ${ENET_LIBS})

# -------- server --------
add_executable(server server.cpp ${STATE_SRC})
target_include_directories(server PRIVATE ${COMMON_INCLUDES})
target_link_libraries(server PRIVATE ${COMMON_LIBS} ${ENET_LIBS})
